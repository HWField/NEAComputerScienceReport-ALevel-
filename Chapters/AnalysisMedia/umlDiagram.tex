
\begin{center}
\begin{tikzpicture}

  \begin{umlpackage}[x=0,y=0]{SimulationSystem}

    % Class: Agent
    \umlclass[x=0, y=0]{Agent}{
      - Name : String \\
      - Health : Float \\
      - Status : String \\
      - Role : Role \\
      - Home : Node \\
      - Workplace : Node \\
      - Location : Node
    }{}

    % Class: Node
    \umlclass[x=6, y=0]{Node}{
      - Type : String \\
      - Capacity : Int \\
      - Location : String \\
      - CurrentOccupants : Agent[]
    }{}

    % Class: Role
    \umlclass[x=6, y=-5]{Role}{
      - Type : String \\
      - StartTime : Time \\
      - LunchBreak : Time \\
      - EndTime : Time \\
      - Priority : Int
    }{}

    % Class: Disease
    \umlclass[x=0, y=-5]{Disease}{
      - Name : String \\
      - DeathRate : Float \\
      - InfectionRate : Float \\
      - Colour : String
    }{}

    % Class: Simulation
    \umlclass[x=3, y=-10]{Simulation}{
      - Agents : Agent[] \\
      - Nodes : Node[] \\
      - Roles : Role[] \\
      - Disease : Disease
    }{}

    % Class: Account
    \umlclass[x=3, y=-15]{Account}{
      - Name : String \\
      - Email : String \\
      - Password : String \\
      - Simulations : Simulation[]
    }{}

    % Associations
    \umluniassoc{Agent}{Node}       % Home
    \umluniassoc{Agent}{Node}       % Workplace
    \umluniassoc{Agent}{Role}
    \umluniassoc{Node}{Agent}    % CurrentOccupants
    \umluniassoc{Simulation}{Agent}
    \umluniassoc{Simulation}{Node}
    \umluniassoc{Simulation}{Role}
    \umluniassoc{Simulation}{Disease}
    \umluniassoc{Account}{Simulation}

  \end{umlpackage}

\end{tikzpicture}
\end{center}
